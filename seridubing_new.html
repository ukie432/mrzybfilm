function parseM3U(text) { const lines = text.split('\n'); const seriesMap = new Map(); let currentPoster = ""; let currentTitle = ""; let currentEpName = ""; for (let i = 0; i < lines.length; i++) { let line = lines[i].trim(); if (line.startsWith("#EXTINF")) { const logoMatch = line.match(/tvg-logo=["']([^"']+)["']/); currentPoster = logoMatch ? logoMatch[1] : ""; const groupMatch = line.match(/group-title=["']([^"']+)["']/); const groupName = groupMatch ? groupMatch[1].trim() : ""; const commaIndex = line.lastIndexOf(','); let rawTitle = commaIndex > -1 ? line.substring(commaIndex + 1).trim() : "Unknown Title"; // JIKA CUMA ADA GROUP "Film" / "Series" ATAU TIDAK ADA GROUP BERSIH: if (!groupName || groupName.toLowerCase() === "film" || groupName.toLowerCase() === "series") { // Deteksi pola S01E01, S1E1, E01, Episode 01, dll. const seasonEpRegex = /(S\d+E\d+|S\d+\s*E\d+|E\d+|Episode\s*\d+)/i; const match = rawTitle.match(seasonEpRegex); if (match) { const splitIndex = match.index; currentTitle = rawTitle.substring(0, splitIndex).trim(); currentEpName = match[0].toUpperCase(); } else if (rawTitle.includes("-")) { let parts = rawTitle.split("-"); currentTitle = parts[0].trim(); currentEpName = parts.slice(1).join("-").trim(); } else { currentTitle = rawTitle.trim(); currentEpName = "Play"; } } else { currentTitle = groupName; currentEpName = rawTitle; } // ======================================================== // PROSES PEMBERSIHAN TITIK PADA JUDUL // ======================================================== // Mengubah semua titik (.) menjadi spasi, lalu merapikan spasi ganda currentTitle = currentTitle.replace(/\./g, ' ').replace(/\s+/g, ' ').trim(); } else if (line.startsWith("http")) { const fallbackPoster = "https://via.placeholder.com/300x400?text=No+Poster"; if (!seriesMap.has(currentTitle)) { seriesMap.set(currentTitle, { title: currentTitle, poster: currentPoster || fallbackPoster, episodes: [] }); } else { let existing = seriesMap.get(currentTitle); if ((!existing.poster || existing.poster === fallbackPoster) && currentPoster) { existing.poster = currentPoster; } } seriesMap.get(currentTitle).episodes.push({ ep: currentEpName, url: line }); } } // AUTO SORT EPISODE (Urut dari S01E01, S01E02, S01E03, dst) const result = Array.from(seriesMap.values()); result.forEach(item => { item.episodes.sort((a, b) => { let numA = a.ep.match(/\d+/g); let numB = b.ep.match(/\d+/g); if (numA && numB) { let lastA = parseInt(numA[numA.length - 1], 10); let lastB = parseInt(numB[numB.length - 1], 10); return lastA - lastB; } return a.ep.localeCompare(b.ep, undefined, { numeric: true, sensitivity: 'base' }); }); }); return result; }
